export const FORBIDDEN_CERTAINTY_WORDS = ['必', '一定', '注定'] as const

## 3.2 ★★ `scoreToTag` 改为 `fortuneTierOf`

```ts
/** 大运/流年：7 档。★ 结构性缺陷封顶（事故 18 的纪律，必须保留并扩展） */
export function fortuneTierOf(score: number, structuralIssues: string[] = []): FortuneTier {
  // ★★ 有结构病者封顶：结构病越多，上限越低
  //    —— 这是事故 18 的修复（"用神到位 ≠ 用神有效"），不得删除
  const cap =
    structuralIssues.length === 0 ? 4 :      // 无结构病 → 可达「大吉」
    structuralIssues.length === 1 ? 2 :      // 1 个     → 最高「小吉」
    0                                        // ≥2 个    → 最高「平」
  const s = Math.min(score, cap)

  if (s >= 4) return '大吉'
  if (s === 3) return '吉'
  if (s === 2) return '小吉'
  if (s >= 0) return '平'          // 0 与 1 合并为「平」（正向不虚增）
  if (s === -1) return '小凶'
  if (s === -2) return '凶'
  return '大凶'                    // ≤ −3
}
```

**★★ 为什么 0 与 1 都归「平」**：
```
· 保证「平」是真正的中间档（不偏乐观）✓
· 且只有【无结构病且多个喜用同现】才够「小吉」以上 ✓
→ ★ 这会让 7 档分布自然偏中，避免"大吉"滥用 ✓（见 §五）
```

**★ 反向不设 floor** ✓（结构病只封上限、不封下限 ✓）
**★★ 但 §五 的断言必须检查下限侧是否被滥用**（事故 19：从"过度乐观"滑到"过度宿命"）。
