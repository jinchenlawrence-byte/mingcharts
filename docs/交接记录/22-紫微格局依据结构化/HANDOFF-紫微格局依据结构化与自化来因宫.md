# 二、步骤 ①：PatternTag 加【依据结构 ＋ 出处】（★ 零风险）

## 2.1 改动

```ts
// 现状（src/lib/ziwei.ts:46）
export interface PatternTag {
  name: string
  level: '吉' | '平' | '忌'
  note: string
}

// ★★★ 扩为（新增字段【全部可选】✗ 以兼容既有构造点 ✓）
export interface PatternTag {
  name: string
  level: '吉' | '平' | '忌'
  note: string
  /** ★ 成立要件：该格局凭什么成立（红线 §8.3-④「点名结构」） */
  required?: string[]
  /** ★ 加分项：使格局更完整/更强 */
  bonus?: string[]
  /** ★ 破格项：使格局减力或转坏 */
  breaking?: string[]
  /** ★ 古籍出处：★ 必须逐条核到书名与篇名 ✗ 不得写"传统说法" */
  source?: string
  /** ★ 依据等级（§8.6.2）：'古籍' | '通行做法' | '产品设定' */
  basisLevel?: '古籍' | '通行做法' | '产品设定'

```

